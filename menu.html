function showCategory(category, element) { // Skrij vse kategorije document.querySelectorAll('.app-list').forEach(list => { list.style.display = 'none'; }); // Pokaži izbrano kategorijo document.getElementById(category).style.display = 'grid'; // Odstrani aktivni razred iz vseh elementov document.querySelectorAll('.navbar a').forEach(a => { a.classList.remove('active'); }); // Dodaj aktivni razred trenutnemu elementu element.classList.add('active'); }
